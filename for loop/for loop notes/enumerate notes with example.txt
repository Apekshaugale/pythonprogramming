s="Hello"  #-----> 0-->H
"""
normal syntax :--->
         
    enumerate(iterable)--->inbuilt function
         |
         |
    Data will convert To object address
         |
         |
    Object Address Data again convert To readable format
         |
         |
    Two ways---->1 -->Typecasting   2--->Looping
         |
         |
    syntax for Typecasting
    
    list(enumerate(iterable))
    tuple(enumerate(iterable))
    dict(enumerate(iterable))
    set(enumerate(iterable))
        |
        |
    syntax for---> Looping
    
    for variable in enumerate(iterable):
        statement
         
         |
         |
   output of enumerate function----->(Position,value)
"""
s="Hello"
print(enumerate(s)) #<enumerate object at 0x000002201988D530>

#way---->1  Typecasting
print(list(enumerate(s)))
#[(0, 'H'), (1, 'e'), (2, 'l'), (3, 'l'), (4, 'o')]

print(tuple(enumerate(s)))
#((0, 'H'), (1, 'e'), (2, 'l'), (3, 'l'), (4, 'o'))

print(set(enumerate(s)))
#{(4, 'o'), (0, 'H'), (2, 'l'), (1, 'e'), (3, 'l')}

print(dict(enumerate(s)))
#{0: 'H', 1: 'e', 2: 'l', 3: 'l', 4: 'o'}

#way--->2
for i in enumerate(s):
    print(i)
"""
(0, 'H')
(1, 'e')
(2, 'l')
(3, 'l')
(4, 'o')

"""
for i,j in enumerate(s):
    print(i,j)
"""
i----->Position
j----->value

Note :--> when we use two reference_variable
          Here will get output in unpacked format
          
"""

# k=[10,20,30,40,50]
